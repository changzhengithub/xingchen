拖拽分为两个部分，拖和放（Drag 和 drop），鼠标拖动某个元素，然后放到某个元素，拖和放都有其对应的方法或事件。我们只要在对应的事件处理相应的逻辑就可以实现拖拽元素了。

# 1、设置元素为可拖放

要拖拽哪个元素需要为其设置draggable属性：

<img draggable="true">

# 2、拖拽事件

拖动的元素触发事件 (源元素):

  • ondragstart - 用户开始拖动元素时触发

  • ondrag - 元素正在拖动时触发

  • ondragend - 用户完成元素拖动后触发

释放到目标元素时触发的事件:

  • ondragenter - 当被鼠标拖动的对象进入其容器范围内时触发此事件

  • ondragover - 当某被拖动的对象在另一对象容器范围内拖动时触发此事件

  • ondragleave - 当被鼠标拖动的对象离开其容器范围内时触发此事件

  • ondrop - 当元素或选中的文本在可释放目标上被释放时触发

拖拽元素要想拖拽到目标元素上，目标元素必须先设置 ondragover 和 ondrop事件才可以，并且drop事件执行需要在dragover事件中阻止默认事件才行。

绑定拖拽事件可以在拖拽元素上直接绑定，也可以通过document进行全局监听。

事件执行过程：ondragstart -> ondrag -> ondragenter -> ondragover -> ondrop/ondragleave -> ondragend

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Document</title>
  </head>
  <body>
    <div class="index">
      <div class="index-drag" draggable="true" ondragstart="dragstart(event)"></div>
      <div class="index-drop"></div>
    </div>


    <script>
      // 拖拽源目标事件
      // 开始拖动
      function dragstart(event) {
        console.log('开始拖动', event)
      }
      // 持续拖动触发
      function drag(event) {
        console.log('持续拖动触发', event)
      }
      // 完成拖动
      function dragend(event) {
        console.log('完成拖动', event)
      }


      // 拖入目标事件
      // 拖入元素
      function dragenter(event) {
        console.log('拖入元素', event)
      }
      // 持续拖入触发
      function dragover(event) {
        console.log('持续拖入触发', event)
        event.preventDefault()
      }
      // 离开拖入元素
      function dragleave(event) {
        console.log('离开拖入元素', event)
      }
      
      // 释放鼠标
      function drop(event) {
        event.preventDefault();
        console.log('释放鼠标', event)
        event.target.appendChild(document.getElementById('drag'));
      }


      // 全局监听可拖拽元素
      document.addEventListener('dragstart', function (event) {
        console.log(event)
      })
      // ...
    </script>
  </body>
</html>

# 3、拖拽传参

从开始拖拽到结束，我们要想知道我们拖拽了哪个元素，我们可以通过 dataTransfer 属性来进行传参。dataTransfer属性上有setData()和getData()两个方法来设置和获取数据，通过拖拽事件event来调用dataTransfer。

// 传入数据
function dragstart(event) {
  event.dataTransfer.setData('keyName', event.target.id)
}
// 获取数据
function drop(event) {
  var data = event.dataTransfer.getData('keyName')
  console.log(data)
}

# 4、自定义拖拽图标

拖拽过程中，浏览器会在鼠标旁显示被拖拽元素的样式。我们可以通过 setDragImage() 方法自定义一张图片。

// 开始拖动
function dragstart(event) {
  // 设置拖拽样式
  var img = new Image();
  img.src = "./imgs/antd.png";
  event.dataTransfer.setDragImage(img, 5, 5);
}

# 5、定义鼠标拖拽效果

// copy/link/move
function dragstart(event) {
  event.dataTransfer.dropEffect = "copy";
}

# 6、拖拽排序

在拖拽时就自动计算拖拽到哪个位置，在鼠标没松开之前就直接让元素插入到对应的位置，元素随着鼠标拖动而跟着替换位置，真实的dom元素已经改变，不需要等到鼠标松开再改变位置。

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Document</title>
    <style>
      .list-item {
        background-color: #266fff;
        border-radius: 5px;
        color: #fff;
        width: 250px;
        padding: 5px;
        margin-bottom: 10px;
      }
      .list-item.moving {
        background-color: transparent;
        color: transparent;
        border: 1px solid #ccc;
      }
    </style>
  </head>
  <body>
    <div class="list">
      <div draggable="true" class="list-item">1</div>
      <div draggable="true" class="list-item">2</div>
      <div draggable="true" class="list-item">3</div>
      <div draggable="true" class="list-item">4</div>
      <div draggable="true" class="list-item">5</div>
    </div>
  </body>
  <script>
    // 拖拽排序在很多站点里面是非常常见的，使用的API就是一个拖拽API


    // 1、让元素变的可拖拽   找到这些元素，给这些元素加上 draggable 属性，值为true，这样就变得可拖拽了
    // 2、拖拽的时候样式得变   class为moving的样式
    const list = document.querySelector('.list')


    // 用事件委托的方式 给父元素绑定事件
    list.addEventListener('dragstart', function (e) {
      // 当拖拽开始的时候要找到拖拽的那个元素给他添加类样式


      // 为什么要用setTimeout,不用的话拖拽的那个元素也会变成虚线（就是添加了类样式后的样子），
      // 它的样式取决于拖拽开始时候元素本身的样式，需要在拖拽开始的时候保持原来的样式 把它变成异步的
      setTimeout(() => {
        e.target.classList.add('moving')
      }, 0)
      sourceNode = e.target
    })
    // 用来记录当前拖拽的是哪个元素
    let sourceNode = null


    // 3、什么时候产生排序？拖拽的时候把拖拽对象放到了某些元素之上，这里就要监听拖拽进入事件
    list.addEventListener('dragenter', function (e) {
      e.preventDefault()
      // 排除掉一些情况，比如拖拽的时候进入了父元素 或者 是本身自己
      if (e.target == list || e.target == sourceNode) return
      const children = [...list.children]
      // 通过所处元素的下标来判断是上方还是下方
      const sourceIndex = children.indexOf(sourceNode)
      const targetIndex = children.indexOf(e.target)
      console.log(sourceIndex, targetIndex)
      // 4、当拖拽的元素进入到别的元素身上的时候，要做的一些事情
      if (sourceIndex < targetIndex) {
        // console.log('下方')
        // 插入到那个元素下一个元素之前
        list.insertBefore(sourceNode, e.target.nextElementSibling)
      } else {
        //  console.log('上方')
        list.insertBefore(sourceNode, e.target)
      }
      console.log(sourceIndex)
    })


    // 拖拽完毕 松开鼠标
    list.addEventListener('dragend', function (e) {
      // 移除掉类样式即可
      sourceNode.classList.remove('moving')
    })
  </script>
</html>
